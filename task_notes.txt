  required_providers {

    }
  }
}

Ex: I need terraform version 1.14.0 and aws version 6.1.0
terraform {
  required_version = "1.14.0"
  required_providers {
    aws = {
      version = "6.1.0"
      source = "hashicorp/aws"
    }
  }
}

Examples
required version = "= 0.13.3" -exact version
required version = "!= 0.13.3" -any version except this
required version = "> 0.13.3" -greater than version
required version = ">= 0.13.3" -greater than or equal
required version = "<= 0.13.3" -less than or equal
required version = "~>= 0.13" -allow Minor level increaments    -- example 0.13.21----> 0.14.0
required version = "~>= 0.13.3" -allow Patch level increaments    -- example 0.13.3----> 0.13.4

Note : By default terraform plan / terraform apply will process all the .tf files and terraform.tfvars file
terraform.tfvars will have the highest priority
To take vaules from other tfvars file we have to define the file name during execution of terraform plan / apply 
(Ex: terraform plan -var-file stage.tfvars)

 variable block - default -- if default is not provided --> prompts during execution
 cmd --> -var var=value (example: terraform plan -var vmsize=t2.small)
 terraform.tfvars --> overrides default values
 *.tfvars --> cmd --> -var-file filename.tfvars (ex: terraform plan -var-file stage.tfvars)
 *.auto.tfvars --> overrides terraform.tfvars

resource "aws_instance" "myserver" {
  ami = "ami-0eb38b817b93460ac"
  instance_type = "t2.micro"
  key_name = "t31key"

    connection {
    type = "ssh"

    host = self.public_ip
    private_key = tls_private_key.vmkeygen.private_key_pem
  }
    source = "deployhttp.sh"
    destination = "/tmp/deployhttp.sh"
    when = create
  }
  provisioner "remote-exec" {
    connection {
    type = "ssh"
    user = "ec2-user"
    host = self.public_ip
    private_key = tls_private_key.vmkeygen.private_key_pem

    inline = [ 
      "chmod +x /tmp/deployhttp.sh",

     ]
  }

    command = "echo The deleted EC2 instance IP address is ${self.public_ip} > deletedvm.txt"
    when = destroy
  }

    Name = "ThahaWebServer"
  }
}
